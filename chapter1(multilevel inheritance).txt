class Person {
    String name;
    int age;

    Person(String name, int age) {
        this.name = name;
        this.age = age;
        System.out.println("Person constructor");
    }

    void display() {
        System.out.println("Name: " + name + ", Age: " + age);
    }
}

class Employee extends Person {
    int employeeId;
    double basicSalary;

    Employee(String name, int age, int employeeId, double basicSalary) {
        super(name, age);
        this.employeeId = employeeId;
        this.basicSalary = basicSalary;
        System.out.println("Employee constructor");
    }

    @Override
    void display() {
        super.display();
        System.out.println("Employee ID: " + employeeId);
        System.out.println("Basic salary: " + basicSalary);
    }
}

class Manager extends Employee {
    String department;
    double bonus;

    Manager(String name, int age, int id, double basic,
            String department, double bonus) {
        super(name, age, id, basic);
        this.department = department;
        this.bonus = bonus;
        System.out.println("Manager constructor");
    }

    double totalSalary() {
        return basicSalary + bonus;
    }

    @Override
    void display() {
        super.display();
        System.out.println("Department: " + department);
        System.out.println("Bonus: " + bonus);
        System.out.println("Total salary: " + totalSalary());
    }
}

public class Main {
    public static void main(String[] args) {
        Manager m = new Manager("Kiran", 35, 101, 60000,
                                "Development", 10000);
        System.out.println("--- Manager details ---");
        m.display();
    }
}
