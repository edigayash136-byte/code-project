class Student {
    int rollNo;
    String name;
    int age;

    void display() {
        System.out.println("Roll No: " + rollNo);
        System.out.println("Name: " + name);
        System.out.println("Age: " + age);
        System.out.println();
    }

    public static void main(String[] args) {
        Student s1 = new Student();
        Student s2 = new Student();

        s1.rollNo = 101;
        s1.name = "Yash";
        s1.age = 19;

        s2.rollNo = 102;
        s2.name = "Rahul";
        s2.age = 20;

        s1.display();
        s2.display();
    }
}
